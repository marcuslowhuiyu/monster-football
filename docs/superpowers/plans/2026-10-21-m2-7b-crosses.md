`CrossResolver.cs`:
```csharp
using MonsterFootball.Simulation.Core;
using MonsterFootball.Simulation.Players;
using MonsterFootball.Simulation.Random;
using MonsterFootball.Simulation.Tactical;

namespace MonsterFootball.Simulation.Actions.Crossing;

/// <summary>
/// Resolves a <see cref="CrossIntent"/> (ADR-036): aimed at the teammate in the opposition penalty area with the most
/// space (largest distance to the nearest opponent; ties → earlier player), else the penalty spot. Two draws scatter the
/// landing point sideways (± 3 m + 0.5 m per Passing point below 10) and lengthways (± 3 m). Flight 20 m/s. Reads state.
/// </summary>
internal sealed class CrossResolver
{
    public const double BaseLateralScatter = 3;
    public const double LateralScatterPerPassing = 0.5;
    public const double LengthScatter = 3;
    public const double BallSpeed = 20;

    /// <summary>Where the cross is aimed (pitch coordinates). Deterministic.</summary>
    public Vec2 Aim(MatchState state, PlayerMatchState crosser)
    {
        var opponents = state.TeamPlayers(crosser.Team.Opponent()).ToList();
        PlayerMatchState? best = null;
        var bestSpace = double.NegativeInfinity;
        foreach (var teammate in state.TeamPlayers(crosser.Team))
        {
            if (teammate == crosser || !state.Zones.IsIn(teammate.Team, teammate.Position, TacticalAreas.OppositionPenaltyArea))
                continue;
            var space = opponents.Count == 0 ? double.MaxValue : opponents.Min(o => o.Position.DistanceTo(teammate.Position));
            if (space > bestSpace)
            {
                best = teammate;
                bestSpace = space;
            }
        }

        return best?.Position ?? state.Pitch.PenaltySpot(crosser.Team.Opponent());
    }

    public CrossResult Resolve(MatchState state, CrossIntent intent, IMatchRandom random)
    {
        var crosser = state.Player(intent.Crosser);
        if (!state.HasBall(intent.Crosser))
            throw new ArgumentException($"{crosser} cannot cross without the ball.", nameof(intent));

        var target = Aim(state, crosser);
        var start = crosser.Position;
        var along = (target - start).Normalized();
        var across = new Vec2(-along.Y, along.X);
        var lateralLimit = BaseLateralScatter + LateralScatterPerPassing * (10 - crosser.Definition.Stats.Passing);
        var lateral = (random.NextDouble() * 2 - 1) * lateralLimit;
        var length = (random.NextDouble() * 2 - 1) * LengthScatter;
        var landing = target + across * lateral + along * length;

        return new CrossResult(crosser.Team, crosser.PlayerId, intent.Kind, start, target, landing, start.DistanceTo(landing) / BallSpeed);
    }
}
```

`AerialContest.cs`:
```csharp
using MonsterFootball.Simulation.Core;
using MonsterFootball.Simulation.Players;
using MonsterFootball.Simulation.Positioning;
using MonsterFootball.Simulation.Random;

namespace MonsterFootball.Simulation.Actions.Crossing;

public enum AerialOutcome
{
    /// <summary>The defending keeper caught it.</summary>
    Claimed,
    AttackerWins,
    DefenderWins,

    /// <summary>Nobody was close enough: the ball lands loose.</summary>
    Loose,
}

/// <summary>
/// Who wins a cross where it lands (ADR-036), among active players within <see cref="ContestRadius"/>: the defending
/// keeper claims it in their goal area; otherwise the best attacker and best defender by aerial rating contest it
/// (P(attacker) = Sigmoid(0.4 × (A − D) − 0.14) ≈ 0.43 between equals, one draw); a side alone wins; nobody → loose.
/// </summary>
internal static class AerialContest
{
    public const double ContestRadius = 3;
    public const double ContestScale = 0.4;
    public const double ContestBias = -0.14;

    public static double AttackerRating(PlayerStats s) => 0.5 * s.Power + 0.3 * s.Awareness + 0.2 * s.Technique;

    public static double DefenderRating(PlayerStats s) => 0.5 * s.Power + 0.3 * s.Awareness + 0.2 * s.Defending;

    public static double AttackerWinChance(PlayerMatchState attacker, PlayerMatchState defender) =>
        Curves.Sigmoid(ContestScale * (AttackerRating(attacker.Definition.Stats) - DefenderRating(defender.Definition.Stats)) + ContestBias);

    public static (AerialOutcome Outcome, PlayerMatchState? Winner) Decide(MatchState state, TeamSide attackers, Vec2 landing, IMatchRandom random)
    {
        var defenders = attackers.Opponent();
        if (InGoalArea(state.Pitch, defenders, landing) && state.Goalkeeper(defenders) is { } keeper
            && keeper.Position.DistanceTo(landing) <= ContestRadius)
            return (AerialOutcome.Claimed, keeper);

        var attacker = Best(state, attackers, landing, AttackerRating);
        var defender = Best(state, defenders, landing, DefenderRating);
        if (attacker is null && defender is null)
            return (AerialOutcome.Loose, null);
        if (defender is null)
            return (AerialOutcome.AttackerWins, attacker);
        if (attacker is null)
            return (AerialOutcome.DefenderWins, defender);
        return random.Roll(AttackerWinChance(attacker, defender))
            ? (AerialOutcome.AttackerWins, attacker)
            : (AerialOutcome.DefenderWins, defender);
    }

    /// <summary>The side's outfield player within reach with the best rating (ties → nearer, then earlier player).</summary>
    private static PlayerMatchState? Best(MatchState state, TeamSide side, Vec2 landing, Func<PlayerStats, double> rating) =>
        state.TeamPlayers(side)
            .Where(p => !p.Role.IsGoalkeeper() && p.Position.DistanceTo(landing) <= ContestRadius)
            .OrderByDescending(p => rating(p.Definition.Stats))
            .ThenBy(p => p.Position.DistanceTo(landing))
            .FirstOrDefault();

    private static bool InGoalArea(PitchGeometry pitch, TeamSide defending, Vec2 point)
    {
        var local = TeamFrame.Convert(pitch, defending, point);
        return local.X >= 0 && local.X <= PitchGeometry.GoalAreaDepth
               && Math.Abs(local.Y - pitch.Width / 2) <= PitchGeometry.GoalAreaWidth / 2;
    }
}
```

```csharp
using MonsterFootball.Simulation.Core;
using MonsterFootball.Simulation.Players;

namespace MonsterFootball.Simulation.Actions.Crossing;

/// <summary>
/// A defender who wins a cross heads it clear (ADR-036): with chance 0.25 behind their own goal line beside the goal (a
/// corner), otherwise 15–25 m away from their goal along the line from its centre through the landing point, rolling.
/// </summary>
internal static class HeaderClearance
{
    public const double BehindChance = 0.25;
    public const double MinDistance = 15;
    public const double MaxDistance = 25;

    public static void Apply(MatchContext context, PlayerMatchState defender, Vec2 landing)
    {
        var state = context.State;
        var pitch = state.Pitch;
        state.RecordTouch(defender.PlayerId);

        if (context.Random.Roll(BehindChance))
        {
            var post = PitchGeometry.GoalWidth / 2 + 1;
            var y = landing.Y >= pitch.Width / 2 ? Math.Max(landing.Y, pitch.Width / 2 + post) : Math.Min(landing.Y, pitch.Width / 2 - post);
            Laws.OutOfPlay.Stop(context, new Vec2(pitch.OwnGoalLineX(defender.Team), y), Laws.BoundaryLine.GoalLine);
            return;
        }

        var away = landing - pitch.OwnGoalCentre(defender.Team);
        var distance = MinDistance + context.Random.NextDouble() * (MaxDistance - MinDistance);
        state.SetBallRolling(landing, LooseBallMotion.Covering(landing, away, distance, state.Clock.Tick));
    }
}
```

```csharp
using System.Collections.Immutable;
using System.Globalization;
using MonsterFootball.Simulation.Actions.Passing;
using MonsterFootball.Simulation.Actions.Shooting;
using MonsterFootball.Simulation.Core;
using MonsterFootball.Simulation.Events;

namespace MonsterFootball.Simulation.Actions.Crossing;

/// <summary>Applies a cross: launch; out of play if its path leaves the pitch; on arrival the aerial contest (ADR-036).</summary>
internal static class CrossExecution
{
    private static readonly ShotResolver Shots = new();

    /// <summary>Launches the cross. Offside is judged as for a pass, except from a corner (Law 11).</summary>
    public static void Launch(MatchContext context, CrossResult result)
    {
        var state = context.State;
        var tick = state.Clock.Tick;
        state.LaunchBall(BallFlight.ForCross(result, tick, tick + BallFlight.TravelTicks(result.TravelDuration, context.Config.TickSeconds)));
        if (result.Kind == CrossKind.OpenPlay && context.Config.OffsideEnabled)
            state.Offside = Laws.Offside.Judge(state, state.Player(result.Crosser));

        context.Emit(MatchEventType.Cross, actor: result.Crosser, team: result.Team, position: result.StartPosition,
            metadata: ImmutableSortedDictionary.CreateRange(
            [
                new KeyValuePair<string, string>("distance", result.Distance.ToString("0.###", CultureInfo.InvariantCulture)),
                new KeyValuePair<string, string>("kind", result.Kind.ToString()),
            ]));
    }

    public static void LeavePlay(MatchContext context, Vec2 crossing, Laws.BoundaryLine line) =>
        Laws.OutOfPlay.Stop(context, crossing, line);

    /// <summary>The contest where the cross lands, then its outcome.</summary>
    public static void Land(MatchContext context, CrossResult result)
    {
        var state = context.State;
        var landing = result.TargetPosition;
        var (outcome, winner) = AerialContest.Decide(state, result.Team, landing, context.Random);
        context.Emit(MatchEventType.CrossContested, actor: winner?.PlayerId, team: winner?.Team ?? result.Team, position: landing,
            metadata: ImmutableSortedDictionary.CreateRange(
            [
                new KeyValuePair<string, string>("outcome", outcome switch
                {
                    AerialOutcome.Claimed => "Claimed",
                    AerialOutcome.AttackerWins => "Shot",
                    AerialOutcome.DefenderWins => "Cleared",
                    _ => "Loose",
                }),
            ]));

        switch (outcome)
        {
            case AerialOutcome.Claimed:
                context.GiveBall(winner!.PlayerId, PossessionCause.Save);
                break;
            case AerialOutcome.AttackerWins:
                if (context.GiveBall(winner!.PlayerId, PossessionCause.Reception)) // false: offside
                    ShotExecution.Strike(context, Shots.ResolveFirstTime(state, new ShotIntent(winner.PlayerId), context.Random));
                break;
            case AerialOutcome.DefenderWins:
                HeaderClearance.Apply(context, winner!, landing);
                break;
            default:
                var travelled = landing - result.StartPosition;
                state.SetBallRolling(landing, LooseBallMotion.Covering(landing, travelled, PassExecution.RollOnFraction * travelled.Length, state.Clock.Tick));
                break;
        }
    }
}
```
